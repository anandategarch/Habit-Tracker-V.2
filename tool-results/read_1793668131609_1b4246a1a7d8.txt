   100→        frame = window.requestAnimationFrame(() => setActiveHabitTimer({ habitId: timer.habitId as string, startedAt: timer.startedAt as number }));

   164→  function toggleHabit(id: string) {
   165→    if (modules.settings.vacationMode) return;
   166→    updateHabits((current) => current.map((habit) => habit.id === id ? { ...habit, completed: !habit.completed } : habit));

   169→  function adjustHabit(id: string, delta: number) {
   170→    if (modules.settings.vacationMode) return;
   171→    updateHabits((current) => current.map((habit) => {
   172→      if (habit.id !== id || habit.kind !== "amount") return habit;
   173→      const currentAmount = Math.max(0, (habit.currentAmount ?? 0) + delta);
   174→      return { ...habit, currentAmount, completed: currentAmount >= (habit.targetAmount ?? 1) };
   175→    }));
   176→  }

   178→  function toggleHabitTimer(id: string) {
   179→    if (modules.settings.vacationMode) return;
   180→    if (activeHabitTimer) {
   181→      const elapsed = Math.max(0, Math.floor((Date.now() - activeHabitTimer.startedAt) / 1000));
   182→      const minutes = Math.floor(elapsed / 60);
   183→      if (minutes > 0) updateHabits((current) => current.map((habit) => habit.id === activeHabitTimer.habitId ? { ...habit, timeSpentMinutes: (habit.timeSpentMinutes ?? 0) + minutes } : habit));
   184→      setActiveHabitTimer(null);
   185→      window.localStorage.removeItem("rutina.v2.active-habit-timer");
   186→      if (activeHabitTimer.habitId === id) return;
   187→    }
   188→    const next = { habitId: id, startedAt: Date.now() };
   189→    setActiveHabitTimer(next);
   190→    window.localStorage.setItem("rutina.v2.active-habit-timer", JSON.stringify(next));
   191→  }

   193→  function submitHabit(event: FormEvent<HTMLFormElement>) {
   194→    event.preventDefault();
   195→    const data = new FormData(event.currentTarget);
   196→    const title = String(data.get("title") ?? "").trim();
   197→    if (!title) return;
   198→
   199→    const scheduleKind = String(data.get("scheduleKind") || "daily");
   200→    let schedule: Habit["schedule"] = { kind: "daily" };
   201→    if (scheduleKind === "weekly") {
   202→      const days = parseNumberList(String(data.get("scheduleDays") || ""), 0, 6);
   203→      if (!days.length) { setToast("Pilih minimal satu hari untuk jadwal mingguan."); return; }
   204→      schedule = { kind: "weekly", days };
   205→    } else if (scheduleKind === "monthly") {
   206→      const dates = parseNumberList(String(data.get("scheduleDates") || ""), 1, 31);
   207→      if (!dates.length) { setToast("Pilih minimal satu tanggal untuk jadwal bulanan."); return; }
   208→      schedule = { kind: "monthly", dates };
   209→    }
   210→
   211→    const category = String(data.get("category") ?? "Personal") as Habit["category"];
   212→    const habit: Habit = {
   213→      id: crypto.randomUUID(),
   214→      title,
   215→      category,
   216→      time: String(data.get("time") ?? "07:00"),
   217→      duration: Math.max(1, Number(data.get("duration")) || 15),
   218→      completed: false,
   219→      icon: category === "Health" ? "movement" : category === "Learning" ? "book" : category === "Mindfulness" ? "mind" : "write",
   220→      streak: 0,
   221→      kind: String(data.get("kind") || "normal") as Habit["kind"],
   222→      targetAmount: Math.max(1, Number(data.get("targetAmount")) || 8),
   223→      currentAmount: 0,
   224→      unit: String(data.get("unit") || "unit").trim().slice(0, 30),
   225→      priority: String(data.get("priority") || "medium") as Habit["priority"],
   226→      difficulty: String(data.get("difficulty") || "easy") as Habit["difficulty"],
   227→      groupName: String(data.get("groupName") || "").trim().slice(0, 60),
   228→      timeSpentMinutes: 0,
   229→      schedule,
   230→    };
   231→
   232→    updateHabits((current) => [...current, habit]);
   233→    setModal(null);
   234→    setToast("Kebiasaan baru berhasil ditambahkan.");

   250→      date: String(data.get("date") || todayISO()),

   253→    };
