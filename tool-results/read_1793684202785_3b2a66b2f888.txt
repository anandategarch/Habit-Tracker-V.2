    65→function GoalsView({ modules, updateModule }: { modules: WorkspaceModules; updateModule: UpdateWorkspaceModule }) {
    66→  function submit(event: FormEvent<HTMLFormElement>) {
    67→    event.preventDefault();
    68→    const data = new FormData(event.currentTarget);
    69→    const title = String(data.get("title") || "").trim();
    70→    if (!title) return;
    71→    const milestones: Milestone[] = String(data.get("milestones") || "").split("\n").map((value) => value.trim()).filter(Boolean).map((value) => ({ id: id(), title: value, completed: false }));
    72→    const goal: Goal = {
    73→      id: id(), title, description: String(data.get("description") || ""),
    74→      category: String(data.get("category") || "Personal"), dueDate: String(data.get("dueDate") || ""),
    75→      milestones, completed: false, createdAt: today(),
    76→    };
    77→    updateModule("goals", (items) => [goal, ...items]);
    78→    event.currentTarget.reset();
    79→  }
    80→  const active = modules.goals.filter((goal) => !goal.completed).length;
    81→  const completed = modules.goals.length - active;
    82→  const steps = modules.goals.reduce((sum, goal) => sum + goal.milestones.length, 0);
    83→  const doneSteps = modules.goals.reduce((sum, goal) => sum + goal.milestones.filter((step) => step.completed).length, 0);
    84→  return <>
    85→    <Intro eyebrow="GOALS & MILESTONES" title="Ubah niat menjadi progres." subtitle="Pecah tujuan besar menjadi langkah yang bisa diselesaikan satu per satu." />

    87→    <div className="module-two-col">
    88→      <Panel title="Tujuan baru" caption="Tuliskan hasil yang ingin dicapai dan milestone-nya">
    89→        <form className="form-grid" onSubmit={submit}>
    90→          <Field label="Nama tujuan"><input className={field} name="title" maxLength={100} required placeholder="Contoh: Menyelesaikan sertifikasi" /></Field>
    91→          <Field label="Deskripsi"><textarea className={field} name="description" rows={2} placeholder="Mengapa tujuan ini penting?" /></Field>
    92→          <div className="form-two-columns">
    93→            <Field label="Kategori"><select className={field} name="category"><option>Personal</option><option>Health</option><option>Learning</option><option>Finance</option><option>Career</option></select></Field>
    94→            <Field label="Tenggat"><input className={field} name="dueDate" type="date" /></Field>
    95→          </div>
    96→          <Field label="Milestones (satu per baris)"><textarea className={field} name="milestones" rows={3} placeholder={"Riset langkah pertama\nSelesaikan bagian awal\nEvaluasi hasil"} /></Field>
    97→          <PrimaryButton type="submit"><Plus size={15} /> Simpan tujuan</PrimaryButton>
    98→        </form>
    99→      </Panel>
   100→      <Panel title="Daftar tujuan" caption={`${modules.goals.length} tujuan tersimpan`}>
   101→        {modules.goals.length ? <div className="module-list goal-list">{modules.goals.map((goal) => {
   102→          const done = goal.milestones.filter((step) => step.completed).length;
   103→          const progress = goal.milestones.length ? Math.round(done / goal.milestones.length * 100) : (goal.completed ? 100 : 0);
   104→          return <article className="module-item goal-card" key={goal.id}>
   105→            <div className="module-item-top"><div><span className="module-tag">{goal.category}</span><h3>{goal.title}</h3><p>{goal.description || "Tidak ada deskripsi."}</p></div><button className="module-icon-action danger" onClick={() => updateModule("goals", (items) => remove(items, goal.id))} aria-label={`Hapus tujuan ${goal.title}`}><Trash2 size={15} /></button></div>
   106→            {goal.dueDate && <small className="module-muted">Tenggat {formatDate(goal.dueDate)}</small>}
   107→            <div className="module-progress-track"><span style={{ width: `${progress}%` }} /></div>
   108→            <div className="module-inline-meta"><small>{done}/{goal.milestones.length} milestone · {progress}%</small><button className={secondary} onClick={() => updateModule("goals", (items) => items.map((item) => item.id === goal.id ? { ...item, completed: !item.completed } : item))}>{goal.completed ? "Buka kembali" : "Tandai selesai"}</button></div>
   109→            {goal.milestones.map((step) => <label className="module-check-row" key={step.id}><input type="checkbox" checked={step.completed} onChange={(event) => updateModule("goals", (items) => items.map((item) => item.id === goal.id ? { ...item, milestones: item.milestones.map((part) => part.id === step.id ? { ...part, completed: event.target.checked } : part) } : item))} /><span className={step.completed ? "is-done" : ""}>{step.title}</span></label>)}
   110→          </article>;
   111→        })}</div> : <NoData>Belum ada tujuan. Mulai dengan satu hasil yang ingin kamu capai.</NoData>}
   112→      </Panel>
   113→    </div>
   114→  </>;

   184→    const next: WellnessEntry = { ...current, weightKg: weight ? Number(weight) : current.weightKg, proteinGrams: Math.max(0, Number(protein) || 0), note };

   195→function TrainingView({ modules, updateModule }: { modules: WorkspaceModules; updateModule: UpdateWorkspaceModule }) {
   196→  const [seconds, setSeconds] = useState(90);
   197→  const [remaining, setRemaining] = useState(90);
   198→  const [running, setRunning] = useState(false);
   199→  const [photoBusy, setPhotoBusy] = useState(false);
   200→  useEffect(() => {
   201→    if (!running) return;
   202→    const timer = window.setInterval(() => setRemaining((value) => {
   203→      if (value <= 1) { window.clearInterval(timer); setRunning(false); return 0; }
   204→      return value - 1;
   205→    }), 1000);
   206→    return () => window.clearInterval(timer);
   207→  }, [running]);
   208→
   209→  function addWorkout(event: FormEvent<HTMLFormElement>) {
   210→    event.preventDefault(); const data = new FormData(event.currentTarget);
   211→    const exercise = String(data.get("exercise") || "").trim(); if (!exercise) return;

   469→async function compressPhoto(file: File): Promise<string> {
   470→  const url = URL.createObjectURL(file);
   471→  try {
   472→    const image = await new Promise<HTMLImageElement>((resolve, reject) => {
   473→      const element = document.createElement("img"); element.onload = () => resolve(element); element.onerror = () => reject(new Error("Gambar tidak dapat dibuka.")); element.src = url;
   474→    });
   475→    const scale = Math.min(1, 480 / Math.max(image.width, image.height));
   476→    const canvas = document.createElement("canvas"); canvas.width = Math.max(1, Math.round(image.width * scale)); canvas.height = Math.max(1, Math.round(image.height * scale));
   477→    const context = canvas.getContext("2d"); if (!context) throw new Error("Browser tidak mendukung kompresi foto.");
   478→    context.drawImage(image, 0, 0, canvas.width, canvas.height);
   479→    let quality = .62; let data = canvas.toDataURL("image/jpeg", quality);
   480→    while (data.length > 260_000 && quality > .3) { quality -= .08; data = canvas.toDataURL("image/jpeg", quality); }
   481→    return data;
   482→  } finally { URL.revokeObjectURL(url); }
